1. List ADT using Arrays and Linked Lists

Displays an array and a linked list, then deletes the value 20 from each.

C++ Source Code

#include<iostream.h>
#include<conio.h>

struct node
{
    int data;
    node *next;
};

void main()
{
    clrscr();

    int arr[10]={10,20,30}, n=3, i;
    node *head=NULL, *p, *temp;

    cout<<"Array List: ";
    for(i=0;i<n;i++)
        cout<<arr[i]<<" ";

    // Array deletion: delete 20
    for(i=1;i<n-1;i++)
        arr[i]=arr[i+1];
    n--;

    cout<<"\\nAfter deletion: ";
    for(i=0;i<n;i++)
        cout<<arr[i]<<" ";

    // Create linked list
    for(i=1;i<=3;i++)
    {
        temp=new node;
        temp->data=i*10;
        temp->next=NULL;

        if(head==NULL)
            head=temp;
        else
            p->next=temp;

        p=temp;
    }

    cout<<"\\nLinked List: ";
    p=head;
    while(p!=NULL)
    {
        cout<<p->data<<" ";
        p=p->next;
    }

    // Linked list deletion: delete 20
    p=head;
    temp=p->next;
    p->next=temp->next;
    delete temp;

    cout<<"\\nAfter deletion: ";
    p=head;
    while(p!=NULL)
    {
        cout<<p->data<<" ";
        p=p->next;
    }

    getch();
}

2. Linear Search

Checks each array element one by one until the required number is found.

C++ Source Code

#include<iostream.h>
#include<conio.h>

void main()
{
    clrscr();
    int a[10], n, i, key;

    cout<<"Enter size: ";
    cin>>n;

    cout<<"Enter numbers: ";
    for(i=0;i<n;i++)
        cin>>a[i];

    cout<<"Enter number to search: ";
    cin>>key;

    for(i=0;i<n;i++)
    {
        if(a[i]==key)
        {
            cout<<"Number found at position "<<i+1;
            getch();
            return;
        }
    }

    cout<<"Number not found";
    getch();
}

3. Stack ADT using a Singly Linked List

Offers Push, Pop, and Display operations. A stack follows LIFO: Last In, First Out.

C++ Source Code

#include<iostream.h>
#include<conio.h>

struct node
{
    int data;
    node *next;
};

void main()
{
    clrscr();
    node *top=NULL, *p;
    int ch, item;

    do
    {
        cout<<"\\n1.Push 2.Pop 3.Display 4.Exit: ";
        cin>>ch;

        if(ch==1)
        {
            cout<<"Enter value: ";
            cin>>item;
            p=new node;
            p->data=item;
            p->next=top;
            top=p;

        {

                top=top->next;

4. Binary Search

Searches a sorted array by repeatedly checking its middle element.

C++ Source Code

#include<iostream.h>
#include<conio.h>

void main()
{
    clrscr();
    int a[10], n, i, key;
    int low, high, mid;

    cout<<"Enter size: ";
    cin>>n;

    cout<<"Enter sorted numbers: ";
    for(i=0;i<n;i++)
        cin>>a[i];

    cout<<"Enter number to search: ";
    cin>>key;

    low=0;
    high=n-1;

    while(low<=high)
    {
        mid=(low+high)/2;

        if(a[mid]==key)
            break;
        else if(key>a[mid])
            low=mid+1;
        else
            high=mid-1;
    }

    if(low<=high)
        cout<<"Number found at position "<<mid+1;
    else
        cout<<"Number not found";

    getch();
}

5. Queue ADT using a Singly Linked List

Offers Insert, Delete, and Display operations. A queue follows FIFO: First In, First Out.

C++ Source Code

#include<iostream.h>
#include<conio.h>

struct node
{
    int data;
    node *next;
};

void main()
{
    clrscr();
    node *front=NULL, *rear=NULL, *p;
    int ch, item;

    do
    {
        cout<<"\\n1.Insert 2.Delete 3.Display 4.Exit: ";
        cin>>ch;

        if(ch==1)
        {
            cout<<"Enter value: ";
            cin>>item;

            p=new node;
            p->data=item;
            p->next=NULL;

            if(front==NULL)
                front=rear=p;
            else
            {
                rear->next=p;
                rear=p;
            }
        }
        else if(ch==2)
        {
            if(front==NULL)
                cout<<"Queue empty";
            else
            {
                cout<<"Deleted: "<<front->data;
                p=front;
                front=front->next;
                delete p;

                if(front==NULL)
                    rear=NULL;
            }
        }
        else if(ch==3)
        {
            for(p=front;p!=NULL;p=p->next)
                cout<<p->data<<" ";
        }
    }while(ch!=4);

    getch();
}

6. Bubble Sort

Compares neighbouring values and swaps them when they are in the wrong order.

C++ Source Code

#include<iostream.h>
#include<conio.h>

void main()
{
    clrscr();
    int a[10], n, i, j, temp;

    cout<<"Enter size: ";
    cin>>n;

    cout<<"Enter numbers: ";
    for(i=0;i<n;i++)
        cin>>a[i];

    for(i=0;i<n-1;i++)
    {
        for(j=0;j<n-i-1;j++)
        {
            if(a[j]>a[j+1])
            {
                temp=a[j];
                a[j]=a[j+1];
                a[j+1]=temp;
            }
        }
    }

    cout<<"Sorted numbers: ";
    for(i=0;i<n;i++)
        cout<<a[i]<<" ";

    getch();
}

Sample Output

Enter size: 5
Enter numbers: 5 2 4 1 3
Sorted numbers: 1 2 3 4 5
7. Breadth-First Search (BFS)

Visits graph vertices level by level, starting at vertex 0.

C++ Source Code

#include<iostream.h>
#include<conio.h>

void main()
{
    clrscr();
    int a[10][10], q[10], visited[10]={0};
    int n, i, j, front=0, rear=0;

    cout<<"Enter number of vertices: ";
    cin>>n;

    cout<<"Enter adjacency matrix:\\n";
    for(i=0;i<n;i++)
        for(j=0;j<n;j++)
            cin>>a[i][j];

    q[rear++]=0;
    visited[0]=1;

    cout<<"BFS: ";

    while(front<rear)
    {
        i=q[front++];
        cout<<i<<" ";

        for(j=0;j<n;j++)
        {
            if(a[i][j]==1 && visited[j]==0)
            {
                q[rear++]=j;
                visited[j]=1;
            }
        }
    }

    getch();
}

8. Selection Sort

Finds the smallest value in the unsorted part and moves it to the next position.

C++ Source Code

#include<iostream.h>
#include<conio.h>

void main()
{
    clrscr();
    int a[10], n, i, j, min, temp;

    cout<<"Enter size: ";
    cin>>n;

    cout<<"Enter numbers: ";
    for(i=0;i<n;i++)
        cin>>a[i];

    for(i=0;i<n-1;i++)
    {
        min=i;

        for(j=i+1;j<n;j++)
            if(a[j]<a[min])
                min=j;

        temp=a[i];
        a[i]=a[min];
        a[min]=temp;
    }

    cout<<"Sorted numbers: ";
    for(i=0;i<n;i++)
        cout<<a[i]<<" ";

    getch();
}

9. Depth-First Search (DFS)

Visits a vertex and explores an unvisited neighbour as deeply as possible before returning.

C++ Source Code

#include<iostream.h>
#include<conio.h>

int a[10][10], visited[10]={0}, n;

void dfs(int v)
{
    int i;
    cout<<v<<" ";
    visited[v]=1;

    for(i=0;i<n;i++)
        if(a[v][i]==1 && visited[i]==0)
            dfs(i);
}

void main()
{
    clrscr();
    int i,j;

    cout<<"Enter number of vertices: ";
    cin>>n;

    cout<<"Enter adjacency matrix:\\n";
    for(i=0;i<n;i++)
        for(j=0;j<n;j++)
            cin>>a[i][j];

    cout<<"DFS: ";
    dfs(0);

    getch();
}

10. Insertion Sort

Takes each value and inserts it into the correct place in the already-sorted part.

C++ Source Code

#include<iostream.h>
#include<conio.h>

void main()
{
    clrscr();
    int a[10], n, i, j, key;

    cout<<"Enter size: ";
    cin>>n;

    cout<<"Enter numbers: ";
    for(i=0;i<n;i++)
        cin>>a[i];

    for(i=1;i<n;i++)
    {
        key=a[i];
        j=i-1;

        while(j>=0 && a[j]>key)
        {
            a[j+1]=a[j];
            j--;
        }

        a[j+1]=key;
    }

    cout<<"Sorted numbers: ";
    for(i=0;i<n;i++)
        cout<<a[i]<<" ";

    getch();
}

Sample Output

Enter size: 5
Enter numbers: 5 2 4 1 3
Sorted numbers: 1 2 3 4 5
